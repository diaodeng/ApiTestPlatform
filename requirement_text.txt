一、Playwright 的“录制”到底是什么？

Playwright 所谓的录制，本质是一个交互式的代码生成器（codegen），而不是一个事件流 SDK。

它的核心能力是：

打开一个真实浏览器（Chromium / Firefox / WebKit）
监听用户在页面上的行为
将行为翻译成 Playwright API 调用
实时展示生成的代码（但不是通过 API 输出）

⚠️ 关键结论先给出：

Playwright 官方只支持「录制 → 生成脚本」模式
❌ 不提供“实时回调 / 事件流 API”给你消费录制过程

五、AST 解析完整实现（核心部分）
1️⃣ 使用工具

2️⃣ 解析入口
import { parse } from '@babel/parser';
import traverse from '@babel/traverse';

export function parsePlaywrightScript(code: string) {
  const ast = parse(code, {
    sourceType: 'module',
    plugins: ['typescript']
  });

  const steps: any[] = [];

  traverse(ast, {
    AwaitExpression(path) {
      const step = parseAwait(path.node);
      if (step) steps.push(step);
    }
  });

  return steps;
}
3️⃣ 解析 await 表达式（关键）
function parseAwait(node) {
  const expr = node.argument;

  if (expr.type !== 'CallExpression') return null;

  return parseAction(expr);
}
4️⃣ 解析 Playwright Action（链式）
示例 AST 结构
page.getByRole(...).click()

是：

CallExpression
 └─ callee: MemberExpression
     └─ object: CallExpression (getByRole)
     └─ property: click
实现解析
function parseAction(call: any) {
  if (call.callee.type !== 'MemberExpression') return null;

  const action = call.callee.property.name;
  const locatorCall = call.callee.object;

  const locator = parseLocator(locatorCall);

  return {
    type: action,
    locator
  };
}
5️⃣ 解析 Locator（核心难点）
function parseLocator(node: any) {
  if (node.type !== 'CallExpression') return null;

  const method = node.callee.property?.name;

  switch (method) {
    case 'getByRole':
      return parseGetByRole(node.arguments);
    case 'getByText':
      return parseGetByText(node.arguments);
    case 'locator':
      return parseLocatorSelector(node.arguments);
  }

  return { raw: 'unsupported' };
}
getByRole 解析
function parseGetByRole(args) {
  const role = args[0].value;
  const options = args[1]?.properties ?? [];

  const result: any = { engine: 'role', role };

  for (const prop of options) {
    if (prop.key.name === 'name') {
      result.name = prop.value.value;
    }
  }

  return result;
}
getByText
function parseGetByText(args) {
  return {
    engine: 'text',
    text: args[0].value
  };
}

2️⃣ Python AST 解析示例（可直接用）
解析入口
import ast

def parse_playwright_script(code: str):
    tree = ast.parse(code)
    steps = []

    for node in ast.walk(tree):
        if isinstance(node, ast.Expr):
            step = parse_expr(node.value)
            if step:
                steps.append(step)

    return steps
解析 click / fill / press
def parse_expr(node):
    if not isinstance(node, ast.Call):
        return None

    if isinstance(node.func, ast.Attribute):
        action = node.func.attr
        locator_call = node.func.value

        locator = parse_locator(locator_call)

        if action == "click":
            return {
                "type": "click",
                "locator": locator
            }

        if action == "fill":
            return {
                "type": "input",
                "locator": locator,
                "value": node.args[0].value
            }

    return None
解析 locator（关键）
def parse_locator(node):
    if not isinstance(node, ast.Call):
        return None

    func = node.func
    if not isinstance(func, ast.Attribute):
        return None

    method = func.attr

    if method == "get_by_role":
        return {
            "engine": "role",
            "role": node.args[0].value,
            **parse_keywords(node.keywords)
        }

    if method == "get_by_text":
        return {
            "engine": "text",
            "text": node.args[0].value
        }

    return {"engine": "raw"}
keyword 参数解析（name / exact）
def parse_keywords(keywords):
    data = {}
    for kw in keywords:
        data[kw.arg] = kw.value.value
    return data
解析 expect（断言）
def parse_expect(node):
    if not isinstance(node.func, ast.Attribute):
        return None

    if node.func.value.func.id != "expect":
        return None

    return {
        "type": "assert",
        "assert": node.func.attr,
        "target": parse_locator(node.func.value.args[0])
    }

(() => {
  if (window.__pw_recorder_installed__) return;
  window.__pw_recorder_installed__ = true;

  function getSelector(el) {
    if (!el || el === document.body) return null;

    // 优先 data-testid
    if (el.dataset && el.dataset.testid) {
      return `[data-testid="${el.dataset.testid}"]`;
    }

    // id
    if (el.id) {
      return `#${el.id}`;
    }

    // role + name（近似 Playwright getByRole）
    const role = el.getAttribute("role");
    const name = el.innerText || el.getAttribute("aria-label");
    if (role && name) {
      return {
        engine: "role",
        role,
        name: name.trim()
      };
    }

    // fallback CSS path
    let path = el.tagName.toLowerCase();
    if (el.className) {
      path += "." + [...el.classList].join(".");
    }
    return path;
  }

  function report(step) {
    try {
      window.__pw_record_step(step);
    } catch (e) {
      console.warn("record step failed", e);
    }
  }

  // click
  document.addEventListener("click", e => {
    const el = e.target;
    report({
      type: "click",
      selector: getSelector(el),
      tag: el.tagName.toLowerCase(),
      text: (el.innerText || "").trim()
    });
  }, true);

  // input / textarea
  document.addEventListener("input", e => {
    const el = e.target;
    if (!["INPUT", "TEXTAREA"].includes(el.tagName)) return;

    report({
      type: "input",
      selector: getSelector(el),
      value: el.value
    });
  }, true);

  // checkbox / radio
  document.addEventListener("change", e => {
    const el = e.target;
    if (el.type === "checkbox" || el.type === "radio") {
      report({
        type: "check",
        selector: getSelector(el),
        checked: el.checked
      });
    }

    if (el.tagName === "SELECT") {
      report({
        type: "select",
        selector: getSelector(el),
        value: el.value
      });
    }
  }, true);

  // key events
  document.addEventListener("keydown", e => {
    if (["Enter", "Tab"].includes(e.key)) {
      report({
        type: "key",
        key: e.key
      });
    }
  }, true);
})();

from typing import Dict, Any

class StepRecorder:
    def __init__(self, send_func):
        self.send_func = send_func

    def on_step(self, step: Dict[str, Any]):
        normalized = self.normalize(step)
        self.send_func(normalized)

    def normalize(self, step):
        """
        JS → 平台 Step DSL
        """
        t = step["type"]

        if t == "click":
            return {
                "action": "click",
                "locator": step["selector"]
            }

        if t == "input":
            return {
                "action": "fill",
                "locator": step["selector"],
                "value": step["value"]
            }

        if t == "check":
            return {
                "action": "check",
                "locator": step["selector"],
                "checked": step["checked"]
            }

        if t == "select":
            return {
                "action": "select",
                "locator": step["selector"],
                "value": step["value"]
            }

        if t == "key":
            return {
                "action": "press",
                "key": step["key"]
            }

        return {
            "action": "raw",
            "data": step
        }

function normalizeText(text) {
  return text.replace(/\s+/g, " ").trim();
}

function isVisible(el) {
  return !!(el.offsetWidth || el.offsetHeight || el.getClientRects().length);
}
2️⃣ Role + Accessible Name（最优）
function getByRoleSelector(el) {
  const role = el.getAttribute("role");
  if (!role) return null;

  let name =
    el.getAttribute("aria-label") ||
    el.getAttribute("aria-labelledby") &&
      document.getElementById(el.getAttribute("aria-labelledby"))?.innerText ||
    el.innerText;

  name = name && normalizeText(name);

  if (!name) return null;

  return {
    engine: "role",
    role,
    name
  };
}
3️⃣ Label（input / textarea / select）
function getByLabelSelector(el) {
  if (!["INPUT", "TEXTAREA", "SELECT"].includes(el.tagName)) return null;

  const id = el.id;
  if (id) {
    const label = document.querySelector(`label[for="${id}"]`);
    if (label) {
      return {
        engine: "label",
        text: normalizeText(label.innerText)
      };
    }
  }

  // 包裹式 label
  let parent = el.parentElement;
  while (parent) {
    if (parent.tagName === "LABEL") {
      return {
        engine: "label",
        text: normalizeText(parent.innerText)
      };
    }
    parent = parent.parentElement;
  }

  return null;
}
4️⃣ Placeholder
function getByPlaceholderSelector(el) {
  const placeholder = el.getAttribute("placeholder");
  if (!placeholder) return null;

  return {
    engine: "placeholder",
    text: normalizeText(placeholder)
  };
}
5️⃣ data-testid（Playwright 强烈推荐）
function getByTestIdSelector(el) {
  const testId =
    el.getAttribute("data-testid") ||
    el.getAttribute("data-test-id") ||
    el.getAttribute("data-test");

  if (!testId) return null;

  return {
    engine: "testid",
    value: testId
  };
}
6️⃣ Text（可见文本）
function getByTextSelector(el) {
  const text = normalizeText(el.innerText || "");
  if (!text || text.length > 80) return null;

  return {
    engine: "text",
    text
  };
}
7️⃣ CSS Fallback（稳定版，不走全路径）
function getCssFallback(el) {
  if (el.id) {
    return {
      engine: "css",
      selector: `#${el.id}`
    };
  }

  let path = el.tagName.toLowerCase();

  if (el.classList.length) {
    path += "." + [...el.classList].slice(0, 2).join(".");
  }

  return {
    engine: "css",
    selector: path
  };
}
8️⃣ 统一入口：Playwright 级 selector 生成器
function generatePlaywrightSelector(el) {
  if (!el || !isVisible(el)) return null;

  return (
    getByRoleSelector(el) ||
    getByLabelSelector(el) ||
    getByPlaceholderSelector(el) ||
    getByTestIdSelector(el) ||
    getByTextSelector(el) ||
    getCssFallback(el)
  );
}

三、为什么“解析 codegen 结果”在平台里会出问题？

我们按 真实工程维度一条条拆。

1️⃣ codegen ≠ 结构化数据，而是“最终脚本”

Playwright codegen 输出的是：

await page.getByRole('button', { name: '登录' }).click();

对平台来说，这里有 5 个问题：
语义混在 JS 语法里
断言、操作、控制流耦合
无法增量更新
无法实时反馈
难以双向编辑

你想要的是：

{
  "action": "click",
  "locator": { ... }
}

而不是：

await page.getByRole(...).click();

脚本是“结果”，不是“模型”

2️⃣ “解析脚本”天然是逆向工程

即使你用 AST（这是唯一正确方式）：

Playwright 可以随时：
改 API 命名
改生成策略

插入 await expect 的组合形式

你就会陷入：

追着 Playwright 内部实现跑

这在平台级产品里是 长期不可控成本。

3️⃣ codegen 不能实时反馈（平台致命点）

你平台需要的通常是：

用户点一下
平台立刻显示一条步骤
可立即编辑 / 删除 / 合并
可立即校验 locator

而 codegen：

只能 全部操作完成后
再拿到 完整脚本
无法“回放到某一步”

这在 UX 上是 根本不可接受的

4️⃣ codegen 断言“很聪明”，但你无法控制

Playwright 自动生成断言：

await expect(page.getByText('登录成功')).toBeVisible();

问题是：

❌ 你不能限制断言类型
❌ 你不能绑定业务语义
❌ 你不能区分“关键断言 / 次要断言”
❌ 你不能做断言模板化

在平台里，断言通常需要：

{
  "type": "assert",
  "level": "critical",
  "retry": 3
}

codegen 根本不认识这些概念

5️⃣ codegen 与平台执行模型不一致（隐性大坑）

平台执行通常是：

Step 1 → 成功 / 失败

自由控制流
await / if / try
expect 可能抛异常

from playwright.sync_api import Page, TimeoutError as PWTimeoutError


class LocatorResolveError(Exception):
    pass


class LocatorResolver:
    def __init__(self, page: Page, timeout: int = 5000):
        self.page = page
        self.timeout = timeout

    def resolve(self, locators: list):
        last_error = None

        for locator in locators:
            try:
                pw_locator = self._build_locator(locator)
                pw_locator.wait_for(state="attached", timeout=self.timeout)
                return pw_locator
            except Exception as e:
                last_error = e

        raise LocatorResolveError(f"All locators failed: {last_error}")

    def _build_locator(self, locator: dict):
        t = locator["type"]

        if t == "role":
            return self.page.get_by_role(
                locator["role"],
                name=locator.get("name")
            )

        if t == "text":
            return self.page.get_by_text(locator["text"], exact=locator.get("exact", False))

        if t == "label":
            return self.page.get_by_label(locator["text"])

        if t == "placeholder":
            return self.page.get_by_placeholder(locator["text"])

        if t == "testid":
            return self.page.get_by_test_id(locator["value"])

        if t == "css":
            return self.page.locator(locator["value"])

        if t == "xpath":
            return self.page.locator(f"xpath={locator['value']}")

        raise ValueError(f"Unsupported locator type: {t}")
2️⃣ ActionExecutor
class ActionExecutor:
    def __init__(self, page: Page):
        self.page = page

    def execute(self, step: dict, locator):
        action = step["action"]

        if action == "click":
            locator.click()
            return

        if action == "fill":
            locator.fill(step["value"])
            return

        if action == "press":
            locator.press(step["key"])
            return

        if action == "check":
            locator.check()
            return

        if action == "uncheck":
            locator.uncheck()
            return

        if action == "hover":
            locator.hover()
            return

        if action == "goto":
            self.page.goto(step["url"])
            return

        if action == "wait":
            self.page.wait_for_timeout(step.get("ms", 1000))
            return

        raise ValueError(f"Unsupported action: {action}")

from playwright.sync_api import expect


class AssertionExecutor:
    def run(self, locator, assertion: dict, page: Page):
        t = assertion["type"]

        if t == "visible":
            expect(locator).to_be_visible()
            return

        if t == "hidden":
            expect(locator).to_be_hidden()
            return

        if t == "enabled":
            expect(locator).to_be_enabled()
            return

        if t == "disabled":
            expect(locator).to_be_disabled()
            return

        if t == "text_equals":
            expect(locator).to_have_text(assertion["value"])
            return

        if t == "text_contains":
            expect(locator).to_contain_text(assertion["value"])
            return

            return

4️⃣ DSL Step Runner（完整流程）
class DSLStepRunner:
    def __init__(self, page: Page):
        self.page = page
        self.action_executor = ActionExecutor(page)
        self.assertion_executor = AssertionExecutor()

    def run_step(self, step: dict):
        resolver = LocatorResolver(
            self.page,
            timeout=step.get("timeout", 5000)
        )

        locator = None
        if "target" in step:
            locator = resolver.resolve(step["target"]["locators"])

        self.action_executor.execute(step, locator)

        for assertion in step.get("assertions", []):
            self.assertion_executor.run(locator, assertion, self.page)

2️⃣ 扩展 LocatorResolver（唯一性 + 排序）
class SmartLocatorResolver:
    BASE_SCORES = {
        "role": 100,
        "label": 95,
        "placeholder": 90,
        "testid": 85,
        "text": 80,
        "css": 60,
        "xpath": 40,
    }

    def __init__(self, page, timeout=5000):
        self.page = page
        self.timeout = timeout

    def resolve(self, locators: list):
        candidates = []

        for locator_def in locators:
            try:
                pw_locator = self._build_locator(locator_def)

                count = pw_locator.count()
                if count == 0:
                    continue

                visible = False
                try:
                    visible = pw_locator.first.is_visible(timeout=1000)
                except Exception:
                    pass

                score = self._calculate_score(locator_def, count, visible)
                candidates.append(
                    LocatorCandidate(
                        locator_def,
                        pw_locator,
                        count,
                        visible,
                        score
                    )
                )

            except Exception:
                continue

        if not candidates:
            raise Exception("No valid locator found")

        # 🔥 核心：自动排序
        candidates.sort(key=lambda x: x.score, reverse=True)

        # 返回最优 locator，但保留诊断信息
        return candidates[0].pw_locator, candidates

    def _calculate_score(self, locator_def, count, visible):
        base = self.BASE_SCORES.get(locator_def["type"], 50)

        if count > 1:
            base -= 30

        if not visible:
            base -= 20

        return base
3️⃣ build_locator（与你现有执行器完全兼容）
    def _build_locator(self, locator):
        t = locator["type"]

        if t == "role":
            return self.page.get_by_role(
                locator["role"],
                name=locator.get("name")
            )

        if t == "text":
            return self.page.get_by_text(
                locator["text"],

        if t == "label":

        if t == "testid":

        if t == "css":

2️⃣ ContextResolver 总入口
class ContextResolver:
    def __init__(self, page, timeout=5000):
        self.page = page
        self.timeout = timeout

    def resolve(self, context_chain: list):
        current = self.page

        for ctx in context_chain:
            if ctx["type"] == "iframe":
                current = self._resolve_iframe(current, ctx)
            elif ctx["type"] == "shadow":
                current = self._resolve_shadow(current, ctx)
            else:
                raise ValueError(f"Unknown context type: {ctx['type']}")

        return current

class IFrameResolver:
    def __init__(self, timeout=5000):
        self.timeout = timeout

    def resolve(self, parent, ctx_def):
        candidates = []

        for locator_def in ctx_def["locators"]:
            try:
                if locator_def["type"] == "css":
                    iframe_el = parent.locator(locator_def["value"])
                elif locator_def["type"] == "name":
                    iframe_el = parent.frame(name=locator_def["value"])
                    if iframe_el:
                        return iframe_el
                    continue
                else:
                    continue

                iframe_el.wait_for(state="attached", timeout=self.timeout)
                frame = iframe_el.content_frame()

                if not frame:
                    continue

                candidates.append(frame)

            except Exception:
                continue

        if not candidates:
            raise Exception("Iframe resolve failed")

        # 默认第一个（你也可以加评分）
        return candidates[0]
接入 ContextResolver
    def _resolve_iframe(self, parent, ctx):
        resolver = IFrameResolver(self.timeout)
        return resolver.resolve(parent, ctx)

class ShadowResolver:
    def __init__(self, timeout=5000):
        self.timeout = timeout

    def resolve(self, parent, ctx_def):
        candidates = []

        for locator_def in ctx_def["locators"]:
            try:
                host = parent.locator(locator_def["value"])
                host.wait_for(state="attached", timeout=self.timeout)

                # shadow 内部上下文
                shadow_root = host.locator(":scope")

                candidates.append(shadow_root)

            except Exception:
                continue

        if not candidates:
            raise Exception("Shadow host resolve failed")

        return candidates[0]

(function () {
  if (window.__pw_highlighter__) return;

  function createBox({ top, left, width, height }, label, color) {
    const box = document.createElement("div");
    box.style.position = "fixed";
    box.style.top = `${top}px`;
    box.style.left = `${left}px`;
    box.style.width = `${width}px`;
    box.style.height = `${height}px`;
    box.style.border = `2px solid ${color}`;
    box.style.zIndex = 2147483647;
    box.style.pointerEvents = "none";
    box.style.boxSizing = "border-box";

    const tag = document.createElement("div");
    tag.textContent = label;
    tag.style.position = "absolute";
    tag.style.top = "-18px";
    tag.style.left = "0";
    tag.style.background = color;
    tag.style.color = "#fff";
    tag.style.fontSize = "12px";
    tag.style.padding = "1px 4px";

    box.appendChild(tag);
    document.body.appendChild(box);

    return box;
  }

  window.__pw_highlighter__ = {
    boxes: [],
    highlight(rect, label, color = "red") {
      const box = createBox(rect, label, color);
      this.boxes.push(box);

    clear() {
      this.boxes.forEach(b => b.remove());
      this.boxes = [];
    }
  };

实际情况
